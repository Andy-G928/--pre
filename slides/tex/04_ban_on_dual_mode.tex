\section{Imitation and Self-Preferencing}

\begin{frame}
  \frametitle{What Is Added?}
  \footnotesize
  This part adds two practices to the baseline dual-mode model: \textbf{product imitation} and \textbf{Self-preferencing}.

  \begin{block}{Product imitation}
    \begin{itemize}
      \item \textbf{Assumption:} copied product gives surplus $\nu+\Delta$.
      \item \textbf{Choice:} $S$: participate or not; then $M$: imitate or not
        if $S$ participates.
      \item \textbf{Motivation:} marketplace data helps $M$ identify and copy
        successful innovations.
    \end{itemize}
  \end{block}

  \begin{block}{Self-preferencing}
    \begin{itemize}
      \item \textbf{Assumption:} consumers discover $S$ through $M$'s recommendation.
      \item \textbf{Choice:} $M$: recommend $S$ or not after prices are set.
      \item \textbf{Motivation:} consumers often rely on platform search, ranking,
        or recommendations to find new products.
    \end{itemize}
  \end{block}

  %   Test a dual-mode ban under perfect imitation and perfect steering.
  % \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Marketplace and Seller Mode}
  \footnotesize
  Marketplace mode: same commission $\tau$ from any marketplace sale.

  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{If $M$ shows $S$}
        Buy $S$ through the marketplace iff:
        \begin{equation}
          \Delta+b-p_i\geq \max\{\Delta-p_o,\ b-\tau,\ 0\}
          \label{eq:show-s-marketplace}
        \end{equation}
        \begin{itemize}
          \item Left: surplus from $S$ on $M$.
          \item Right: other best outside option.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{If $M$ hides $S$}
        Buy a fringe seller on the marketplace iff:
        \begin{equation}
          b-\tau\geq 0
          \label{eq:hide-s-fringe}
        \end{equation}
        \begin{itemize}
          \item Left: surplus from a fringe seller on $M$.
          \item Right: no purchase.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \begin{block}{Recommendation rule}
    \begin{itemize}
      \item Show $S$ if Eq. \eqref{eq:show-s-marketplace} holds: $S$ creates a marketplace sale, so $M$ earns commission $\tau$ with show $S$.
      \item Hide $S$ if Eq. \eqref{eq:show-s-marketplace} fails but Eq. \eqref{eq:hide-s-fringe} holds: hiding $S$ redirects consumers to fringe sellers, so $M$ still earns commission $\tau$.
    \end{itemize}
  \end{block}

  Seller mode: $M$ sells only its own product; no marketplace listing for $S$ (unchanged vs baseline).
\end{frame}

\begin{frame}
  \frametitle{Dual Mode: Recommendation Decision}
  % \footnotesize

  \begin{block}{What does $M$ compare?}
    \begin{itemize}
      \item Own expected margin from selling $M$'s product or imitation.
      \item Expected commission from showing $S$.
    \end{itemize}
  \end{block}

  \begin{block}{Recommendation rule}
    \begin{itemize}
    \item $M$ choose to hide $S$
    \begin{itemize}
      \item $M$'s expected margin $>$ expected commission from $S$;
      \item Or consumers prefer $M$ to fringe sellers (Eq. \eqref{eq:show-s-marketplace} fails).
    \end{itemize}
    \item $M$ choose to show $S$ 
    \begin{itemize}
      \item $M$'s expected margin $<$ expected commission from $S$;
      \item and consumers prefer $S$ to fringe sellers (Eq. \eqref{eq:show-s-marketplace} holds).
    \end{itemize}
  \end{itemize}
  \end{block}

\end{frame}


\subsection{}
\begin{frame}
  \frametitle{Pricing Without Imitation}
  \small
  If $M$ does not imitate, steering creates an exploitative equilibrium.

  \begin{block}{Exploitative equilibrium}
    $M$ does not show $S$ and sells to all consumers:
    \begin{equation}
      p_m^*=\min\{\tau,b\}+\sigma .
    \end{equation}
  \end{block}

  \begin{equation}
    \Pi^{\text{exploit}}_{no-imi}
    =
    \bigl(\min\{\tau,b\}+\sigma\bigr)
    G\bigl(\nu+b-\min\{\tau,b\}\bigr).
  \end{equation}

  \begin{alertblock}{Intuition}
    Steering lets $M$ avoid direct price competition with the innovative seller.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Pricing With Imitation}
  \footnotesize
  Imitation matters when $S$'s product is better than $M$'s original product:
  \begin{equation}
    \Delta>\sigma .
  \end{equation}

  \begin{block}{Exploitative equilibrium with imitation}
    $M$ sells the copied product to all consumers at
    \begin{equation}
      p_m^*=\min\{\tau,b\}+\Delta .
    \end{equation}
  \end{block}

  \begin{block}{Price-squeeze equilibrium with imitation}
    If it exists, homogeneous-product competition implies
    \begin{equation}
      p_i^*=p_m^*=\tau .
    \end{equation}
  \end{block}

  \begin{alertblock}{Mechanism}
    Imitation extracts innovation surplus; steering prevents $S$ from using
    discovery to discipline the platform.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Proposition 5: Dual Mode Equilibrium}
  \small
  With product imitation and self-preferencing, $S$ always chooses minimum innovation:
  \begin{equation}
    \Delta^{dual}=\Delta^l .
  \end{equation}

  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Case & Equilibrium conduct & $M$'s profit \\
      \midrule
      $\sigma\geq\Delta^l$
        & no imitation; $M$ sells
        & $(b+\sigma)G(\nu)$ \\
      $\sigma<\Delta^l$
        & high fee or imitation
        & $(b+\Delta^l)G(\nu)$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Summary}
    \begin{equation}
      \Pi^{dual}=(b+\max\{\sigma,\Delta^l\})G(\nu).
    \end{equation}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Why Innovation Collapses}
  \small
  In every relevant dual-mode configuration, $S$'s equilibrium operating profit is zero.

  \begin{itemize}
    \item If $M$ imitates, the copied product captures the innovation surplus.
    \item If $M$ does not imitate because $\sigma\geq\Delta$, $M$ already has the better product.
    \item If $M$ does not imitate because $\tau$ is high, the commission extracts $S$'s margin.
  \end{itemize}

  \begin{equation}
    \pi_S(\Delta)=-K(\Delta)
    \quad\Rightarrow\quad
    \Delta=\Delta^l .
  \end{equation}

  \begin{alertblock}{Economic intuition}
    Imitation and steering remove both rewards from innovation: product advantage
    and access to consumers.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Proposition 6: Ban Dual Mode}
  \small
  Because dual mode is weakly best for $M$, a ban forces $M$ into a pure mode.

  \begin{center}
    \scriptsize
    \begin{tabular}{llllll}
      \toprule
      Region & New mode & $\Pi$ & $\pi$ & CS & $\Delta,W$ \\
      \midrule
      $\sigma\geq\Delta^l$
        & Seller & $=$ & $=$ & $=$ & $\Delta=,\ W=$ \\
      $0<\sigma<\Delta^l$
        & Seller & $\downarrow$ & $=$ & $=$ & $\Delta=,\ W\downarrow$ \\
      $\sigma\leq0$
        & Marketplace & $\downarrow$ & $\uparrow$ & $=$ & $\Delta\uparrow,\ W\uparrow$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Difference from the baseline}
    With perfect steering, consumer surplus is already low in dual mode, so
    banning dual mode no longer changes CS.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Why the Dual-Mode Ban Can Still Lower Welfare}
  \small
  The empirically relevant case is often $0<\sigma<\Delta^l$.

  \begin{itemize}
    \item $M$ has a selling advantage relative to fringe sellers: $\sigma>0$.
    \item $S$'s innovative product is still superior: $\Delta^l>\sigma$.
    \item After the ban, $M$ switches to seller mode.
    \item Consumers cannot combine $S$'s better product with the platform benefit $b$.
  \end{itemize}

  \begin{alertblock}{Policy lesson}
    A structural ban does not restore innovation or effective competition; it
    only removes the marketplace channel from $M$'s choice set.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Behavioral Remedies}
  \small
  Section 4 then compares three conduct remedies with the structural ban.

  \begin{center}
    \begin{tabular}{lll}
      \toprule
      Remedy & Direct target & Restored mechanism \\
      \midrule
      Ban self-preferencing & display choice & on-platform competition \\
      Ban imitation & copying choice & innovation incentives \\
      Ban both & display and copying & competition and innovation \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Main question}
    Can targeted conduct rules preserve the useful parts of dual mode while
    removing the harmful practices?
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Ban Self-Preferencing Only}
  \small
  $M$ must always show $S$ when $S$ is listed.

  \begin{itemize}
    \item The exploitative no-display equilibrium disappears.
    \item The showrooming constraint on $\tau$ is restored.
    \item Product imitation is still allowed.
  \end{itemize}

  \begin{block}{Post-intervention dual mode}
    \begin{equation}
      \tau=b,\qquad \Delta=\Delta^l,\qquad \Pi=(b+\max\{\sigma-\Delta^l,0\})G(\nu+\Delta^l),
      \qquad \pi=0 .
    \end{equation}
  \end{block}

  \begin{alertblock}{Intuition}
    Display neutrality restores price competition, but copying still destroys
    $S$'s innovation incentive.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Proposition 7: Effects of Banning Self-Preferencing}
  \small
  There is a cutoff
  \begin{equation}
    \underline{\sigma}^{steer}\in(0,\min\{\Delta^l,\underline{\sigma}\}]
  \end{equation}
  such that:

  \begin{center}
    \scriptsize
    \begin{tabular}{llllll}
      \toprule
      Region & $M$'s mode & $\Pi$ & $\pi$ & CS & $\Delta,W$ \\
      \midrule
      $\sigma\geq\Delta^l$
        & Seller & $=$ & $=$ & $=$ & $\Delta=,\ W=$ \\
      $\underline{\sigma}^{steer}<\sigma<\Delta^l$
        & Seller & $\downarrow$ & $=$ & $=$ & $\Delta=,\ W\downarrow$ \\
      $\sigma\leq\underline{\sigma}^{steer}$
        & Dual & $\downarrow$ & $=$ & $\uparrow$ & $\Delta=,\ W\uparrow$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Interpretation}
    If $M$ remains in dual mode, consumers gain from lower prices and higher demand.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Ban Imitation Only}
  \small
  Banning imitation is equivalent to letting $M$ credibly commit not to copy.

  \begin{block}{Innovation-restoring condition}
    If
    \begin{equation}
      b+\max\{\sigma,\Delta^l\}\leq\bar{\tau}
    \end{equation}
    or
    \begin{equation}
      \bar{\tau}G(\nu+\sigma+b)
      \geq
      (b+\max\{\sigma,\Delta^l\})G(\nu),
    \end{equation}
    then $M$ sets $\tau=\bar{\tau}$ and $S$ chooses $\Delta=\bar{\Delta}$.
  \end{block}

  \begin{alertblock}{Intuition}
    A no-copying commitment can benefit $M$ when larger innovation expands demand
    enough to offset a less extractive commission.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Proposition 8: Effects of Banning Imitation}
  \small
  $M$ always continues in dual mode after this remedy.

  \begin{center}
    \scriptsize
    \begin{tabular}{llllll}
      \toprule
      Condition & $\Pi$ & $\pi$ & CS & $\Delta$ & $W$ \\
      \midrule
      $b+\max\{\sigma,\Delta^l\}>\bar{\tau}$ and (14) fails
        & $=$ & $=$ & $=$ & $=$ & $=$ \\
      $b+\max\{\sigma,\Delta^l\}\leq\bar{\tau}$ or (14) holds
        & $\uparrow$ & $=$ & $\uparrow$ & $\uparrow$ & $\uparrow$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Mechanism}
    When the ban matters, equilibrium moves from $\Delta^l$ to $\bar{\Delta}$ and
    demand rises from $G(\nu)$ to $G(\nu+\sigma+b)$.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Ban Both Practices}
  \small
  If imitation and self-preferencing are both banned, the dual-mode equilibrium
  becomes the baseline dual-mode equilibrium from Proposition 3.

  \begin{itemize}
    \item Showing $S$ restores on-platform price competition.
    \item No-copying restores $S$'s innovation incentives.
    \item The showrooming constraint again limits $\tau$.
  \end{itemize}

  \begin{block}{Synergy}
    The condition for innovation to increase is less restrictive than under a
    ban on imitation alone, because self-preferencing no longer weakens the
    showrooming constraint.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Proposition 9: Effects of Banning Both}
  \small
  The post-remedy mode depends on the same cutoff $\underline{\sigma}$ from the baseline.

  \begin{center}
    \scriptsize
    \begin{tabular}{llllll}
      \toprule
      Region & $M$'s mode & $\Pi$ & $\pi$ & CS & $\Delta,W$ \\
      \midrule
      $\sigma\geq\max\{\underline{\sigma},\Delta^l\}$
        & Seller & $=$ & $=$ & $=$ & $\Delta=,\ W=$ \\
      $\underline{\sigma}<\sigma<\Delta^l$
        & Seller & $\downarrow$ & $=$ & $=$ & $\Delta=,\ W\downarrow$ \\
      $\sigma\leq\underline{\sigma}$
        & Dual & $\downarrow$ & $\uparrow$ if $b<\bar{\tau}$ & $\uparrow$
        & $\Delta\uparrow$ if $b<\bar{\tau}$ or (8); $W\uparrow$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{alertblock}{Takeaway}
    When $M$ stays in dual mode, banning both practices raises consumer surplus
    and welfare by restoring the baseline dual-mode mechanisms.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{A Data-Sharing Alternative}
  \small
  The paper also discusses a possible rule requiring $M$ to share third-party
  seller data with all third parties.

  \begin{itemize}
    \item This removes $M$'s exclusive imitation advantage.
    \item But now all fringe sellers can imitate $S$.
    \item $S$'s innovation incentive is again eliminated.
  \end{itemize}

  \begin{block}{Comparison}
    In the model, data sharing is weakly worse than banning both imitation and
    self-preferencing because it does not restore $S$'s incentive to innovate.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Corollary 2: Behavioral Remedies vs. Dual-Mode Ban}
  \small
  First consider $\sigma>0$, so a dual-mode ban makes $M$ choose seller mode.

  \begin{center}
    \scriptsize
    \begin{tabular}{llllll}
      \toprule
      Remedy & Relevant region & $\Pi$ & $\pi$ & CS & $\Delta,W$ \\
      \midrule
      Ban self-preferencing
        & $\sigma\leq\underline{\sigma}^{steer}$ & $\uparrow$ & $=$ & $\uparrow$ & $\Delta=,\ W\uparrow$ \\
      Ban self-preferencing
        & $\sigma>\underline{\sigma}^{steer}$ & $=$ & $=$ & $=$ & $\Delta=,\ W=$ \\
      Ban imitation
        & all $\sigma>0$ & $\uparrow$ & $=$ & $\uparrow$ & $\Delta\uparrow,\ W\uparrow$ \\
      Ban both
        & $\sigma\leq\underline{\sigma}$ & $\uparrow$ & $\uparrow$ & $\uparrow$ & $\Delta\uparrow,\ W\uparrow$ \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{block}{Message for empirically relevant cases}
    Conduct remedies weakly dominate the structural ban, and often strictly improve outcomes.
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Corollary 2 When $\sigma\leq0$}
  \small
  If $\sigma\leq0$, a dual-mode ban makes $M$ choose marketplace mode.

  \begin{center}
    \scriptsize
    \begin{tabular}{llll}
      \toprule
      Remedy & Always weakly higher & Innovation & Welfare \\
      \midrule
      Ban self-preferencing
        & $\pi$, CS
        & lower than marketplace
        & up iff condition (9) holds \\
      Ban imitation
        & $\pi$, CS
        & up iff $b+\Delta^l\leq\bar{\tau}$ or (14)
        & up iff $\Delta$ rises \\
      Ban both
        & $\pi$, CS
        & up iff $b\leq\bar{\tau}$ or (8)
        & up iff $\Delta$ rises or (9) holds \\
      \bottomrule
    \end{tabular}
  \end{center}

  \begin{alertblock}{Why ambiguity returns}
    When fringe sellers have the selling advantage, marketplace mode can be a
    strong benchmark; conduct remedies need enough innovation or demand gains to dominate it.
  \end{alertblock}
\end{frame}

\begin{frame}
  \frametitle{Section 4 Takeaways}
  \small
  \begin{enumerate}
    \item Perfect imitation and perfect steering make dual mode weakly best for $M$.
    \item In dual mode, $S$ chooses $\Delta^l$ because innovation rents are extracted.
    \item A dual-mode ban does not directly fix copying or steering.
    \item Banning self-preferencing restores price competition but not innovation.
    \item Banning imitation restores innovation when commitment is valuable enough.
    \item Banning both practices restores the baseline dual-mode mechanisms.
  \end{enumerate}

  \begin{alertblock}{Main policy message}
    The harmful conduct is often a better regulatory target than the platform's
    dual role itself.
  \end{alertblock}
\end{frame}
